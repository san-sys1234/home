function toggleTask(x){
  const wasComplete=plannedToday().length>0 && plannedToday().every(isDone);
  // Daily routines are independent calendar-day occurrences.
  if(isDailyTask(x)){

    if(isDone(x)){
      unmarkDone(target);delete state.lastDone[lastKey(target)];

  save();render();

}
